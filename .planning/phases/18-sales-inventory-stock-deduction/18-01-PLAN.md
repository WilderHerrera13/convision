# Plan 18-01: Wire inventory repos into sale service; deductStock on Create, revertStock on Cancel

## Objective

Add `domain.InventoryItemRepository` and `domain.StockMovementRepository` to `internal/sale/service.go` so that every sale creation deducts stock for `Product.tracks_stock=true` items and writes an `exit` StockMovement in the kardex, and every sale cancellation restores that stock with an `adjustment_add` StockMovement. Sales always complete regardless of stock level. Items with `tracks_stock=false` (lenses) are skipped entirely. No HTTP endpoints or frontend code changes.

## must_haves

- [ ] `Service` struct in `internal/sale/service.go` has fields `itemRepo domain.InventoryItemRepository` and `movementRepo domain.StockMovementRepository`
- [ ] `NewService` function signature accepts `itemRepo domain.InventoryItemRepository` and `movementRepo domain.StockMovementRepository` as the last two parameters before `logger`
- [ ] Private method `deductStock(ctx context.Context, saleID uint, branchID uint, items []domain.SaleItem, userID uint)` exists in `service.go`
- [ ] Private method `revertStock(ctx context.Context, saleID uint, branchID uint, items []domain.SaleItem)` exists in `service.go`
- [ ] `Create()` calls `s.deductStock(...)` after `s.saleRepo.Create(s.db, sale)` succeeds
- [ ] `Cancel()` calls `s.revertStock(...)` before changing sale status
- [ ] `StockMovementRepository` interface in `domain/inventory.go` includes `FindBySaleAndProduct(db *gorm.DB, saleID, productID uint) (*StockMovement, error)`
- [ ] `StockMovementRepository` postgres implementation has `FindBySaleAndProduct` method
- [ ] `cmd/api/main.go` passes `inventoryItemRepo, stockMovementRepo` to `salesvc.NewService` before `logger`
- [ ] `make build` exits 0

```go
func (r *StockMovementRepository) FindBySaleAndProduct(db *gorm.DB, saleID, productID uint) (*domain.StockMovement, error) {
    var m domain.StockMovement
    refType := domain.ReferenceTypeSale
    err := db.
        Where("reference_type = ? AND reference_id = ? AND product_id = ? AND movement_type = ?",
            refType, saleID, productID, domain.MovementTypeExit).
        Order("created_at DESC").
        First(&m).Error
    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return nil, &domain.ErrNotFound{Resource: "stock_movement"}
        }
        return nil, err
    }
    return &m, nil
}
```

**`deductStock` — called after a sale is persisted:**

```go
// deductStock reduces InventoryItem.Quantity for each SaleItem whose Product.tracks_stock=true
// and records a StockMovement of type exit. Always best-effort: warns on missing stock, never
// blocks the sale.
func (s *Service) deductStock(ctx context.Context, saleID uint, branchID uint, items []domain.SaleItem, userID uint) {
    for _, item := range items {
        if item.ProductID == nil {
            continue
        }

        product, err := s.productRepo.GetByID(s.db, *item.ProductID)
        if err != nil {
            s.logger.Warn("deductStock: product not found",
                zap.Uint("product_id", *item.ProductID),
                zap.Uint("sale_id", saleID),
                zap.Error(err),
            )
            continue
        }
        if !product.TracksStock {
            continue // lens or catalog-only product — no deduction
        }

        // Find ALL inventory items for this (product, branch); select the one with highest quantity (D-02).
        invItems, _, err := s.itemRepo.List(s.db, map[string]any{
            "product_id": *item.ProductID,
            "branch_id":  branchID,
        }, 1, 100) // fetch all warehouses, pick best
        if err != nil || len(invItems) == 0 {
            s.logger.Warn("deductStock: no inventory item found, skipping",
                zap.Uint("product_id", *item.ProductID),
                zap.Uint("branch_id", branchID),
                zap.Uint("sale_id", saleID),
            )
            continue
        }

        // Select warehouse with highest quantity.
        best := invItems[0]
        for _, inv := range invItems[1:] {
            if inv.Quantity > best.Quantity {
                best = inv
            }
        }
        invItem := &best

        requested := item.Quantity
        available := invItem.Quantity

        var deducted int
        if available <= 0 {
            s.logger.Warn("deductStock: zero stock, sale proceeds without deduction",
                zap.Uint("product_id", *item.ProductID),
                zap.Uint("inventory_item_id", invItem.ID),
                zap.Int("requested", requested),
                zap.Uint("sale_id", saleID),
            )
            deducted = 0
        } else if available < requested {
            s.logger.Warn("deductStock: insufficient stock, deducting available quantity only",
                zap.Uint("product_id", *item.ProductID),
                zap.Uint("inventory_item_id", invItem.ID),
                zap.Int("available", available),
                zap.Int("requested", requested),
                zap.Uint("sale_id", saleID),
            )
            deducted = available
        } else {
            deducted = requested
        }

        if deducted == 0 {
            continue // nothing to write
        }

        // Wrap update + movement creation in a transaction so they are atomic.
        // Best-effort: log warning on failure but do not block the sale.
        txProductID := *item.ProductID
        txDeducted := deducted
        txUserID := userID
        txSaleID := saleID
        txInvItem := *invItem
        txErr := s.db.Transaction(func(tx *gorm.DB) error {
            txInvItem.Quantity = txInvItem.Quantity - txDeducted
            if err := s.itemRepo.Update(tx, &txInvItem); err != nil {
                return err
            }
            refType := domain.ReferenceTypeSale
            movement := &domain.StockMovement{
                ProductID:      txProductID,
                WarehouseID:    txInvItem.WarehouseID,
                MovementType:   domain.MovementTypeExit,
                ReferenceType:  &refType,
                ReferenceID:    &txSaleID,
                QuantityBefore: invItem.Quantity,
                QuantityDelta:  -txDeducted,
                QuantityAfter:  txInvItem.Quantity,
                PerformedBy:    &txUserID,
                Notes:          "",
            }
            return s.movementRepo.Create(tx, movement)
        })
        if txErr != nil {
            s.logger.Warn("deductStock: failed to record stock deduction in kardex",
                zap.Uint("product_id", *item.ProductID),
                zap.Uint("sale_id", saleID),
                zap.Error(txErr),
            )
        }
    }
}

// revertStock restores InventoryItem.Quantity for each SaleItem whose Product.tracks_stock=true
// by looking up the original exit movement recorded at sale creation. Always best-effort.
func (s *Service) revertStock(ctx context.Context, saleID uint, branchID uint, items []domain.SaleItem) {
    for _, item := range items {
        if item.ProductID == nil {
            continue
        }

        product, err := s.productRepo.GetByID(s.db, *item.ProductID)
        if err != nil {
            s.logger.Warn("revertStock: product not found",
                zap.Uint("product_id", *item.ProductID),
                zap.Uint("sale_id", saleID),
                zap.Error(err),
            )
            continue
        }
        if !product.TracksStock {
            continue
        }

        // Find the original exit movement to know how much was actually deducted.
        origMovement, err := s.movementRepo.FindBySaleAndProduct(s.db, saleID, *item.ProductID)
        if err != nil {
            s.logger.Warn("revertStock: no original exit movement found, skipping",
                zap.Uint("product_id", *item.ProductID),
                zap.Uint("sale_id", saleID),
            )
            continue
        }

        restoredQty := -origMovement.QuantityDelta // QuantityDelta was negative for exit
        if restoredQty <= 0 {
            continue
        }

        // Find the current inventory item.
        invItems, _, err := s.itemRepo.List(s.db, map[string]any{
            "product_id": *item.ProductID,
            "branch_id":  branchID,
        }, 1, 1)
        if err != nil || len(invItems) == 0 {
            s.logger.Warn("revertStock: inventory item not found, skipping restoration",
                zap.Uint("product_id", *item.ProductID),
                zap.Uint("sale_id", saleID),
            )
            continue
        }

        txInvItem := *invItems[0]
        txRestoredQty := restoredQty
        txProductID := *item.ProductID
        txSaleID := saleID
        quantityBefore := txInvItem.Quantity

        // Wrap update + movement creation in a transaction so they are atomic.
        // Best-effort: log warning on failure but do not block the cancellation.
        txErr := s.db.Transaction(func(tx *gorm.DB) error {
            txInvItem.Quantity = quantityBefore + txRestoredQty
            if err := s.itemRepo.Update(tx, &txInvItem); err != nil {
                return err
            }
            refType := domain.ReferenceTypeSale
            movement := &domain.StockMovement{
                ProductID:      txProductID,
                WarehouseID:    txInvItem.WarehouseID,
                MovementType:   domain.MovementTypeAdjustmentAdd,
                ReferenceType:  &refType,
                ReferenceID:    &txSaleID,
                QuantityBefore: quantityBefore,
                QuantityDelta:  txRestoredQty,
                QuantityAfter:  txInvItem.Quantity,
                Notes:          "stock reversal due to sale cancellation",
            }
            return s.movementRepo.Create(tx, movement)
        })
        if txErr != nil {
            s.logger.Warn("revertStock: failed to record stock reversal in kardex",
                zap.Uint("product_id", *item.ProductID),
                zap.Uint("sale_id", saleID),
                zap.Error(txErr),
            )
        }
    }
}
```

Note: `revertStock` sets `PerformedBy` to nil intentionally — the original creator is not available without loading the full sale. The `Notes` field identifies the reversal unambiguously.
</action>

```go
func (s *Service) Cancel(id uint) (*domain.Sale, error) {
    sale, err := s.saleRepo.GetByID(s.db, id)
    if err != nil {
        return nil, err
    }

    s.revertStock(context.Background(), sale.ID, sale.BranchID, sale.Items)

    sale.Status = domain.SaleStatusCancelled
    if err := s.saleRepo.Update(s.db, sale); err != nil {
        return nil, err
    }
    return s.saleRepo.GetByID(s.db, id)
}
```

`revertStock` is called before setting the status to `cancelled` so that if the status update fails (very unlikely), the stock was still returned — consistent with best-effort semantics.
