```go
// TestPermissionIntersection_WithRestrictions verifies that when the optica
// has restrictions, only the intersection of role permissions and allowed keys
// ends up in the JWT permissions.
func TestPermissionIntersection_WithRestrictions(t *testing.T) {
	// Setup: role has 3 permissions, optica allows only 1
	rolePerms := []string{"patients:view", "patients:edit", "appointments:view"}
	allowedKeys := []string{"patients:view"}

	filtered := applyPermissionCeiling(rolePerms, allowedKeys, true, "optica_schema")

	if len(filtered) != 1 {
		t.Fatalf("expected 1 permission after intersection, got %d: %v", len(filtered), filtered)
	}
	if filtered[0] != "patients:view" {
		t.Errorf("expected patients:view, got %s", filtered[0])
	}
}

// TestPermissionIntersection_NoRestrictions verifies that when the optica has
// no restriction rows, all role permissions pass through unchanged.
func TestPermissionIntersection_NoRestrictions(t *testing.T) {
	rolePerms := []string{"patients:view", "patients:edit", "appointments:view"}

	filtered := applyPermissionCeiling(rolePerms, nil, false, "optica_schema")

	if len(filtered) != 3 {
		t.Fatalf("expected 3 permissions (no restriction), got %d", len(filtered))
	}
}

// TestPermissionIntersection_SuperAdmin verifies that super admin schema
// bypasses the intersection entirely.
func TestPermissionIntersection_SuperAdmin(t *testing.T) {
	rolePerms := []string{"patients:view", "patients:edit"}
	allowedKeys := []string{"patients:view"} // would restrict if applied

	filtered := applyPermissionCeiling(rolePerms, allowedKeys, true, "platform")

	// Super admin: no intersection applied
	if len(filtered) != 2 {
		t.Fatalf("expected 2 permissions for super admin (no ceiling), got %d", len(filtered))
	}
}

// applyPermissionCeiling extracts and tests the intersection logic.
// If the auth service exposes this as a standalone function, call it directly.
// Otherwise, reproduce the logic here for unit testing.
func applyPermissionCeiling(rolePerms []string, allowedKeys []string, hasRestriction bool, schemaName string) []string {
	if schemaName == "platform" {
		return rolePerms
	}
	if !hasRestriction {
		return rolePerms
	}
	allowedSet := make(map[string]struct{}, len(allowedKeys))
	for _, k := range allowedKeys {
		allowedSet[k] = struct{}{}
	}
	filtered := rolePerms[:0]
	for _, p := range rolePerms {
		if _, ok := allowedSet[p]; ok {
			filtered = append(filtered, p)
		}
	}
	return filtered
}
```

If the auth service was refactored to expose a standalone `intersectPermissions(rolePerms []string, allowed []string, hasRestriction bool, schemaName string) []string` function, import and call it directly instead of duplicating. Adapt the test to the actual function signature.

Also check if an `authsvc` import is needed — only import if you're testing the actual service struct integration, not just the logic helper.
