# Plan 21-06: Laboratory Filters — special _search and _assigned_uid keys

## Objective

The laboratory module uses two special pseudo-keys injected directly by the handler — `_search` (ILIKE on order number and patient name) and `_assigned_uid` (subquery on status notes). These need to become explicit named fields on typed Filter structs. The frontend currently sends `s_f/s_v/s_o=or` to the laboratory list endpoint; after this migration the backend will silently ignore those params (ShouldBindQuery ignores unknown keys) while correctly processing the typed fields.

    ```go
    func (r *LaboratoryRepository) ListOrders(db *gorm.DB, f domain.LaboratoryOrderFilter) ([]*domain.LaboratoryOrder, int64, error) {
    	f.Clamp()
    	var orders []*domain.LaboratoryOrder
    	var total int64

    	q := db.Model(&domain.LaboratoryOrder{})
    	if f.BranchID != nil {
    		q = q.Where("laboratory_orders.branch_id = ?", *f.BranchID)
    	}
    	if f.PatientID != nil {
    		q = q.Where("patient_id = ?", *f.PatientID)
    	}
    	if f.LaboratoryID != nil {
    		q = q.Where("laboratory_id = ?", *f.LaboratoryID)
    	}
    	if f.Status != "" {
    		q = q.Where("laboratory_orders.status = ?", f.Status)
    	}
    	if f.Priority != "" {
    		q = q.Where("priority = ?", f.Priority)
    	}
    	// f.Search replaces the _search special key
    	if f.Search != "" {
    		like := "%" + f.Search + "%"
    		q = q.Where(
    			"order_number ILIKE ? OR EXISTS (SELECT 1 FROM patients p WHERE p.id = laboratory_orders.patient_id AND (p.first_name ILIKE ? OR p.last_name ILIKE ?))",
    			like, like, like,
    		)
    	}
    	// f.AssignedUID replaces the _assigned_uid special key
    	if f.AssignedUID != nil {
    		pattern := fmt.Sprintf("%%[uid:%d]%%", *f.AssignedUID)
    		q = q.Where(
    			"EXISTS (SELECT 1 FROM laboratory_order_statuses los WHERE los.laboratory_order_id = laboratory_orders.id AND los.notes LIKE ?)",
    			pattern,
    		)
    	}

    	if err := q.Count(&total).Error; err != nil {
    		return nil, 0, err
    	}
    	err := q.Offset(f.Offset()).Limit(f.PerPage).Order("laboratory_orders.id desc").Find(&orders).Error
    	return orders, total, err
    }
    ```
