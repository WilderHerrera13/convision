# Plan 21-02: Simple Stateless Domains — Role and Notification filters

## Objective

Migrate the Role and Notification list endpoints from ad-hoc `c.Query()` calls to typed `ShouldBindQuery` Filter structs. Both domains have one or two filter fields, no s_f/s_v usage, and no branch injection complexity — ideal first migration targets.


    Replace with:
    ```go
    func (r *RoleRepository) List(db *gorm.DB, f domain.RoleFilter) ([]*domain.Role, int64, error) {
    	f.Clamp()
    	var roles []*domain.Role
    	var total int64

    	q := db.Model(&domain.Role{})
    	if f.Name != "" {
    		q = q.Where("name ILIKE ?", "%"+f.Name+"%")
    	}

    	if err := q.Count(&total).Error; err != nil {
    		return nil, 0, err
    	}

    	err := q.Offset(f.Offset()).Limit(f.PerPage).Order("id asc").Find(&roles).Error
    	if err != nil {
    		return nil, 0, err
    	}
    	return roles, total, nil
    }
    ```

    ```go
    func (r *NotificationRepository) List(db *gorm.DB, f domain.NotificationFilter) ([]*domain.Notification, int64, error) {
    	f.Clamp()
    	var items []*domain.Notification
    	var total int64

    	q := db.Model(&domain.Notification{})
    	if f.Archived != nil {
    		q = q.Where("archived = ?", *f.Archived)
    	}
    	if f.Unread != nil {
    		if *f.Unread {
    			q = q.Where("read_at IS NULL")
    		} else {
    			q = q.Where("read_at IS NOT NULL")
    		}
    	}

    	if err := q.Count(&total).Error; err != nil {
    		return nil, 0, err
    	}
    	err := q.Offset(f.Offset()).Limit(f.PerPage).Order("id desc").Find(&items).Error
    	return items, total, err
    }
    ```
