```ts
getLensCatalog: async (params: LensCatalogParams = {}) => {
  const { page = 1, perPage = 15, search, brand_id, supplier_id, status } = params;
  const query = new URLSearchParams({ page: String(page), per_page: String(perPage) });
  if (search) query.set('search', search);
  if (brand_id) query.set('brand_id', String(brand_id));
  if (supplier_id) query.set('supplier_id', String(supplier_id));
  if (status) query.set('status', status);
  const response = await api.get<{ current_page: number; data: LensCatalogItem[]; last_page: number; per_page: number; total: number }>(
    `/inventory/lens-catalog?${query.toString()}`
  );
  return response.data;
},

getStockMovements: async (params: { page?: number; perPage?: number; product_id?: number; warehouse_id?: number; movement_type?: string } = {}) => {
  const { page = 1, perPage = 20, product_id, warehouse_id, movement_type } = params;
  const query = new URLSearchParams({ page: String(page), per_page: String(perPage) });
  if (product_id) query.set('product_id', String(product_id));
  if (warehouse_id) query.set('warehouse_id', String(warehouse_id));
  if (movement_type) query.set('movement_type', movement_type);
  const response = await api.get<{ current_page: number; data: StockMovement[]; last_page: number; per_page: number; total: number }>(
    `/inventory/movements?${query.toString()}`
  );
  return response.data;
},

createAdjustment: async (data: CreateAdjustmentParams) => {
  const response = await api.post<InventoryAdjustment>('/inventory/adjustments', data);
  return response.data;
},

approveAdjustment: async (id: number) => {
  const response = await api.patch<InventoryAdjustment>(`/inventory/adjustments/${id}/approve`);
  return response.data;
},

rejectAdjustment: async (id: number, notes?: string) => {
  const response = await api.patch<InventoryAdjustment>(`/inventory/adjustments/${id}/reject`, { notes });
  return response.data;
},
```
