# Plan 13-03 — Lens-as-Product Unification

## Goal
Re-route lens bulk import to create `Product` + `ProductLensAttributes` records (with `product_type='lens'` and `tracks_stock=false`) instead of `Lens` records. Add a `GET /v1/inventory/lens-catalog` endpoint that returns products where `product_type = 'lens'` paginated, with support for prescription filters. func (r *ProductRepository) ListLensCatalog(filters map[string]any, page, perPage int) ([]*domain.Product, int64, error) {
```go
	var data []*domain.Product
	var total int64

	q := r.db.Model(&domain.Product{}).
		Where("products.product_type = ?", domain.ProductTypeLens).
		Preload("Brand").
		Preload("LensAttributes").
		Preload("LensAttributes.LensType").
		Preload("LensAttributes.Material").
		Preload("LensAttributes.LensClass").
		Preload("LensAttributes.Treatment")

	if v, ok := filters["brand_id"]; ok {
		q = q.Where("products.brand_id = ?", v)
	}
	if v, ok := filters["supplier_id"]; ok {
		q = q.Where("products.supplier_id = ?", v)
	}
	if v, ok := filters["status"]; ok {
		q = q.Where("products.status = ?", v)
	}
	if v, ok := filters["search"]; ok && v != "" {
		q = q.Where("products.internal_code ILIKE ? OR products.identifier ILIKE ?",
			"%"+v.(string)+"%", "%"+v.(string)+"%")
	}

	// Prescription range filters via join to product_lens_attributes
	prescriptionKeys := []string{"sphere_od", "cylinder_od", "addition_od", "sphere_os", "cylinder_os", "addition_os"}
	hasPrescription := false
	for _, k := range prescriptionKeys {
		if _, ok := filters[k]; ok {
			hasPrescription = true
			break
		}
	}
	if hasPrescription {
		q = q.Joins("JOIN product_lens_attributes pla ON pla.product_id = products.id")
		if v, ok := filters["sphere_od"]; ok {
			q = q.Where("pla.sphere_min <= ? AND pla.sphere_max >= ?", v, v)
		}
		if v, ok := filters["cylinder_od"]; ok {
			q = q.Where("pla.cylinder_min <= ? AND pla.cylinder_max >= ?", v, v)
		}
		if v, ok := filters["addition_od"]; ok {
			q = q.Where("pla.addition_min <= ? AND pla.addition_max >= ?", v, v)
		}
	}

	if err := q.Count(&total).Error; err != nil {
		return nil, 0, err
	}

	offset := (page - 1) * perPage
	err := q.Order("products.internal_code ASC").
		Offset(offset).Limit(perPage).Find(&data).Error
	return data, total, err
}
```

Note: The `LensAttributes` preload path must match the field name in `domain.Product` — verify the field is `LensAttributes *ProductLensAttributes gorm:"foreignKey:ProductID"`. If the field is named `LensAttributes`, use `"LensAttributes"`.
</action>

```go
func (s *Service) ListLensCatalog(filters map[string]any, page, perPage int) (*LensCatalogListOutput, error) {
	if page < 1 {

	if int(total)%perPage != 0 {

	}
	if lastPage < 1 {
		lastPage = 1
	}
	return &LensCatalogListOutput{
		CurrentPage: page,

		LastPage:    lastPage,
		PerPage:     perPage,
```

```go
// ListLensCatalog returns the paginated lens product catalog (product_type = 'lens').
func (h *Handler) ListLensCatalog(c *gin.Context) {
	page, perPage := parsePagination(c)
	filters := map[string]any{}
	if v := c.Query("brand_id"); v != "" {
		if id, err := strconv.Atoi(v); err == nil {
			filters["brand_id"] = id
		}
	}
	if v := c.Query("supplier_id"); v != "" {
		if id, err := strconv.Atoi(v); err == nil {
			filters["supplier_id"] = id
		}
	}
	if v := c.Query("status"); v != "" {
		filters["status"] = v
	}
	if v := c.Query("search"); v != "" {
		filters["search"] = v
	}
	for _, key := range []string{"sphere_od", "cylinder_od", "addition_od", "sphere_os", "cylinder_os", "addition_os"} {
		if v := c.Query(key); v != "" {
			if f, err := strconv.ParseFloat(v, 64); err == nil {
				filters[key] = f
			}
		}
	}
	out, err := h.product.ListLensCatalog(filters, page, perPage)
	if err != nil {
		respondError(c, err)
		return
	}
	c.JSON(http.StatusOK, out)
}
```
