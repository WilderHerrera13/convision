```python
import pytest
from playwright.sync_api import Page, expect

VIEWPORTS = {
    'iphone_se': {'width': 375, 'height': 667},
    'iphone_14': {'width': 390, 'height': 844},
    'ipad_portrait': {'width': 768, 'height': 1024},
    'ipad_landscape': {'width': 1024, 'height': 768},
    'laptop': {'width': 1280, 'height': 800},
    'desktop': {'width': 1440, 'height': 900},
}

PAGES_TO_TEST = [
    '/login',
    '/admin/dashboard',
    '/admin/patients',
    '/receptionist/appointments/new',
    '/receptionist/sales/new',
    '/admin/laboratory-orders/new',
]

@pytest.mark.parametrize('viewport_name,viewport', VIEWPORTS.items())
@pytest.mark.parametrize('path', PAGES_TO_TEST)
def test_page_no_horizontal_overflow(page: Page, viewport_name: str, viewport: dict, path: str):
    page.set_viewport_size(viewport)
    page.goto(f'http://localhost:4300{path}')
    page.wait_for_load_state('networkidle')

    body_width = page.evaluate('() => document.body.scrollWidth')
    viewport_width = page.evaluate('() => window.innerWidth')

    assert body_width <= viewport_width, (
        f'Horizontal overflow at {path} on {viewport_name}: '
        f'body scrollWidth={body_width} > viewport={viewport_width}'
    )

@pytest.mark.parametrize('viewport_name,viewport', VIEWPORTS.items())
def test_login_responsive(page: Page, viewport_name: str, viewport: dict):
    page.set_viewport_size(viewport)
    page.goto('http://localhost:4300/login')
    page.wait_for_load_state('networkidle')

    login_form = page.locator('form')
    expect(login_form).to_be_visible()

    if viewport['width'] < 640:
        expect(page.locator('img[alt="Convision"]').first).not_to_be_visible()
```

3. Run the tests: `cd qa-automation && python -m pytest tests/test_responsive.py -v --headed` (note: tests will need the app running on port 4300)
