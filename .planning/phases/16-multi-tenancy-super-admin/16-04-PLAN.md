# 16-04 — Super Admin API

## Goal

Create `OpticaService` with Create (two-phase: INSERT optica → CREATE SCHEMA → run migrations → seed admin+branch → seed flags), List, Get, Update, Deactivate. Create `FeatureFlagService` with List, BulkUpdate, SingleToggle, with cache invalidation. Add handlers and routes under `/api/v1/super-admin/` protected by `RequireRole(super_admin)`.

## Must Haves (goal-backward verification)

- `POST /api/v1/super-admin/opticas` creates an optica, provisions its schema, seeds admin user + branch + feature flags
- `GET /api/v1/super-admin/opticas` lists all opticas (paginated)
- `PATCH /api/v1/super-admin/opticas/:id` updates optica name/plan/active status
- Deactivated optica rejects tenant login (403 "óptica inactiva")
- `PUT /api/v1/super-admin/opticas/:id/features` bulk updates flags and invalidates cache
- Reserved slugs (`admin`, `platform`, etc.) rejected on create
- Idempotent: duplicate slug returns 409 Conflict

type Service struct {
	repo         domain.OpticaRepository
	featureRepo  domain.OpticaFeatureRepository
	featureCache *featurecache.Cache
	opticaCache  *opticacache.Cache
	db           *gorm.DB
	logger       *zap.Logger

func NewService(
	repo domain.OpticaRepository,
	featureRepo domain.OpticaFeatureRepository,
	featureCache *featurecache.Cache,
	opticaCache *opticacache.Cache,
	db *gorm.DB,
	logger *zap.Logger,
) *Service {
	return &Service{
		repo: repo, featureRepo: featureRepo,
		featureCache: featureCache, opticaCache: opticaCache,
		db: db, logger: logger,
	}
}

type CreateOpticaInput struct {
	Name     string               `json:"name"     binding:"required,min=2,max=150"`
	Slug     string               `json:"slug"     binding:"required,min=2,max=60"`
	Plan     string               `json:"plan"     binding:"required,oneof=standard premium enterprise"`
	Admin    CreateTenantAdminInput `json:"admin"    binding:"required"`
}

type CreateTenantAdminInput struct {
	Name     string `json:"name"     binding:"required,min=2,max=150"`
	Email    string `json:"email"    binding:"required,email"`
	Password string `json:"password" binding:"required,min=6"`
}

func (s *Service) Create(input CreateOpticaInput) (*domain.Optica, error) {
	if domain.IsReservedSlug(input.Slug) {
		return nil, &domain.ErrReservedSlug{Slug: input.Slug}
	}
	schemaName := "optica_" + input.Slug
	optica := &domain.Optica{
		Slug: input.Slug, Name: input.Name,
		Plan: input.Plan, IsActive: true, SchemaName: schemaName,
	}
	if err := s.repo.Create(optica); err != nil {
		return nil, err
	}
	if err := s.db.Exec("CREATE SCHEMA " + schemaName).Error; err != nil {
		s.repo.Delete(optica.ID)
		return nil, &domain.ErrSchemaCreationFailed{Schema: schemaName, Err: err}
	}
	err := s.db.Transaction(func(tx *gorm.DB) error {
		tx.Exec("SET LOCAL search_path = ?", schemaName)
		if err := s.runMigrations(tx, schemaName); err != nil {
			return fmt.Errorf("migrations: %w", err)
		}
		if err := s.seedTenant(tx, input.Admin, input.Slug); err != nil {
			return fmt.Errorf("seed: %w", err)
		}
		return nil
	})
	if err != nil {
		s.db.Exec("DROP SCHEMA IF EXISTS " + schemaName + " CASCADE")
		s.repo.Delete(optica.ID)
		return nil, err
	}
	if fErr := s.featureRepo.SeedDefaults(optica.ID); fErr != nil {
		s.logger.Error("failed to seed feature flags for optica",
			zap.Uint("optica_id", optica.ID), zap.Error(fErr))
	}
	s.opticaCache.Upsert(&opticacache.Entry{
		ID: optica.ID, Slug: optica.Slug,
		SchemaName: optica.SchemaName, IsActive: optica.IsActive,
	})
	s.logger.Info("optica created", zap.Uint("id", optica.ID), zap.String("slug", optica.Slug))
	return optica, nil
}

func (s *Service) List(page, perPage int) ([]*domain.Optica, int64, error) {
	return s.repo.List(page, perPage)
}

func (s *Service) GetByID(id uint) (*domain.Optica, error) {
	return s.repo.GetByID(id)
}

func (s *Service) Update(id uint, input UpdateOpticaInput) (*domain.Optica, error) {
	optica, err := s.repo.GetByID(id)
	if err != nil {
		return nil, err
	}
	if input.Name != nil {
		optica.Name = *input.Name
	}
	if input.Plan != nil {
		optica.Plan = *input.Plan
	}
	if input.IsActive != nil {
		optica.IsActive = *input.IsActive
	}
	if err := s.repo.Update(optica); err != nil {
		return nil, err
	}
	s.opticaCache.Upsert(&opticacache.Entry{
		ID: optica.ID, Slug: optica.Slug,
		SchemaName: optica.SchemaName, IsActive: optica.IsActive,
	})
	return optica, nil
}

type UpdateOpticaInput struct {
	Name     *string `json:"name"`
	Plan     *string `json:"plan"`
	IsActive *bool   `json:"is_active"`
}

func (s *Service) seedTenant(tx *gorm.DB, admin CreateTenantAdminInput, slug string) error {
	hash, err := bcrypt.GenerateFromPassword([]byte(admin.Password), bcrypt.DefaultCost)
	if err != nil {
		return err
	}
	if err := tx.Exec(`INSERT INTO users (name, email, password_hash, role, active)
		VALUES (?, ?, ?, 'admin', true)`, admin.Name, admin.Email, string(hash)).Error; err != nil {
		return err
	}
	if err := tx.Exec(`INSERT INTO branches (name, city, is_primary)
		VALUES ('Sede Principal', '', true)`).Error; err != nil {
		return err
	}
	return nil
}

func (s *Service) runMigrations(tx *gorm.DB, schemaName string) error {
	files, err := tenantMigrations.ReadDir("db/migrations/platform")
	if err != nil {
		return err
	}
	var names []string
	for _, f := range files {
		if !f.IsDir() {
			names = append(names, f.Name())
		}
	}
	sort.Strings(names)
	for _, name := range names {
		content, err := tenantMigrations.ReadFile("db/migrations/platform/" + name)
		if err != nil {
			return fmt.Errorf("read %s: %w", name, err)
		}
		if err := tx.Exec(string(content)).Error; err != nil {
			return fmt.Errorf("exec %s: %w", name, err)
		}
	}
	return nil
}

type FeatureService struct {

	logger  *zap.Logger
}

func NewFeatureService(
	repo domain.OpticaFeatureRepository,
	cache *featurecache.Cache,
	logger *zap.Logger,
) *FeatureService {
	return &FeatureService{repo: repo, cache: cache, logger: logger}
}

func (s *FeatureService) List(opticaID uint) ([]*domain.OpticaFeature, error) {
	return s.repo.ListByOpticaID(opticaID)
}

func (s *FeatureService) BulkUpdate(opticaID uint, toggles []domain.FeatureToggle) error {
	if err := s.repo.BulkUpsert(opticaID, toggles); err != nil {
		return err
	}
	s.cache.Invalidate(opticaID)
	s.logger.Info("feature flags updated", zap.Uint("optica_id", opticaID))
	return nil
}

func (s *FeatureService) Toggle(opticaID uint, featureKey string, isEnabled bool) error {
	if err := s.repo.Toggle(opticaID, featureKey, isEnabled); err != nil {
		return err
	}
	s.cache.Invalidate(opticaID)
	return nil
}

func (s *FeatureService) AllKeys() []string {
	return domain.AllFeatureKeys
}

package v1

import (
	"net/http"
	"strconv"

	"github.com/gin-gonic/gin"

	"github.com/convision/api/internal/domain"
	opticasvc "github.com/convision/api/internal/optica"
)

type OpticaResource struct {
	ID         uint   `json:"id"`
	Slug       string `json:"slug"`
	Name       string `json:"name"`
	Plan       string `json:"plan"`
	IsActive   bool   `json:"is_active"`
	SchemaName string `json:"schema_name"`
	CreatedAt  string `json:"created_at"`
}

func toOpticaResource(o *domain.Optica) OpticaResource {
	return OpticaResource{
		ID: o.ID, Slug: o.Slug, Name: o.Name, Plan: o.Plan,
		IsActive: o.IsActive, SchemaName: o.SchemaName,
		CreatedAt: o.CreatedAt.UTC().Format(timeFormat) + "Z",
	}
}

func (h *Handler) ListOpticas(c *gin.Context) {
	page, _ := strconv.Atoi(c.DefaultQuery("page", "1"))
	perPage, _ := strconv.Atoi(c.DefaultQuery("per_page", "15"))
	opticas, total, err := h.optica.List(page, perPage)
	if err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"message": "internal server error"})
		return
	}
	data := make([]OpticaResource, len(opticas))
	for i, o := range opticas {
		data[i] = toOpticaResource(o)
	}
	c.JSON(http.StatusOK, gin.H{
		"data": data, "total": total,
		"current_page": page, "per_page": perPage,
	})
}

func (h *Handler) CreateOptica(c *gin.Context) {
	var input opticasvc.CreateOpticaInput
	if err := c.ShouldBindJSON(&input); err != nil {
		c.JSON(http.StatusUnprocessableEntity, gin.H{"message": err.Error()})
		return
	}
	optica, err := h.optica.Create(input)
	if err != nil {
		var reserved *domain.ErrReservedSlug
		if errors.As(err, &reserved) {
			c.JSON(http.StatusUnprocessableEntity, gin.H{"message": reserved.Error()})
			return
		}
		respondError(c, err)
		return
	}
	c.JSON(http.StatusCreated, toOpticaResource(optica))
}

func (h *Handler) GetOptica(c *gin.Context) {
	id, err := parseID(c, "id")
	if err != nil {
		return
	}
	o, err := h.optica.GetByID(id)
	if err != nil {
		respondError(c, err)
		return
	}
	c.JSON(http.StatusOK, toOpticaResource(o))
}

func (h *Handler) UpdateOptica(c *gin.Context) {
	id, err := parseID(c, "id")
	if err != nil {
		return
	}
	var input opticasvc.UpdateOpticaInput
	if err := c.ShouldBindJSON(&input); err != nil {
		c.JSON(http.StatusUnprocessableEntity, gin.H{"message": err.Error()})
		return
	}
	o, err := h.optica.Update(id, input)
	if err != nil {
		respondError(c, err)
		return
	}
	c.JSON(http.StatusOK, toOpticaResource(o))
}

package v1

import (
	"net/http"

	"github.com/gin-gonic/gin"

	"github.com/convision/api/internal/domain"
)

type FeatureFlagResource struct {
	FeatureKey string `json:"feature_key"`
	IsEnabled  bool   `json:"is_enabled"`
}

func (h *Handler) ListOpticaFeatures(c *gin.Context) {
	opticaID, err := parseID(c, "id")
	if err != nil {
		return
	}
	features, err := h.featureFlag.List(opticaID)
	if err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"message": "internal server error"})
		return
	}
	data := make([]FeatureFlagResource, len(features))
	for i, f := range features {
		data[i] = FeatureFlagResource{FeatureKey: f.FeatureKey, IsEnabled: f.IsEnabled}
	}
	c.JSON(http.StatusOK, gin.H{"features": data})
}

func (h *Handler) BulkUpdateOpticaFeatures(c *gin.Context) {
	opticaID, err := parseID(c, "id")
	if err != nil {
		return
	}
	var body struct {
		Features []domain.FeatureToggle `json:"features" binding:"required"`
	}
	if err := c.ShouldBindJSON(&body); err != nil {
		c.JSON(http.StatusUnprocessableEntity, gin.H{"message": err.Error()})
		return
	}
	if err := h.featureFlag.BulkUpdate(opticaID, body.Features); err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"message": "internal server error"})
		return
	}
	c.JSON(http.StatusOK, gin.H{"message": "Banderas actualizadas correctamente"})
}

func (h *Handler) ToggleOpticaFeature(c *gin.Context) {
	opticaID, err := parseID(c, "id")
	if err != nil {
		return
	}
	featureKey := c.Param("key")
	var body struct {
		IsEnabled bool `json:"is_enabled"`
	}
	if err := c.ShouldBindJSON(&body); err != nil {
		c.JSON(http.StatusUnprocessableEntity, gin.H{"message": err.Error()})
		return
	}
	if err := h.featureFlag.Toggle(opticaID, featureKey, body.IsEnabled); err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"message": "internal server error"})
		return
	}
	c.JSON(http.StatusOK, gin.H{"message": "Bandera actualizada"})
}

func (h *Handler) ListFeatureKeys(c *gin.Context) {
	keys := h.featureFlag.AllKeys()
	c.JSON(http.StatusOK, gin.H{"feature_keys": keys})
}
